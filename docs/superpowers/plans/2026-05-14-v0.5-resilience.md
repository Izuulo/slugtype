> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make slugtype tolerate mid-test terminal shrinks (pause + overlay + resume), display dim text on terminals without 24-bit color, and bring the README in line with what the app does.

**Architecture:** Three pure helpers (`is_too_small`, `apply_pause_offset`, `draw_too_small_overlay`) plus a curses-bound pause loop (`_wait_until_restored`). The pause loop shifts `started_at` forward by the pause duration so `TestState` keeps its current shape. Color fallback gets a three-way branch (truecolor / 16-color / 8-color). README gets text edits.

**Tech Stack:** Python 3.9+ stdlib only (`curses`, `dataclasses`, `os`, `time`). Tests are pure-Python pytest (no TTY required).

**Spec:** `docs/superpowers/specs/2026-05-14-v0.5-resilience-design.md`

```python
def apply_pause_offset(state: TestState, pause_seconds: float) -> TestState:
    """Credit `pause_seconds` back to the runner by shifting started_at
    forward, so time spent on the 'too small' overlay doesn't count toward
    the test. No-op when the test hasn't started or has already finished.

    Returns the same instance (not a copy) on the no-op paths so callers
    can use `out is state` to detect "nothing changed".
    """
    if state.started_at is None or state.finished:
        return state
    return replace(state, started_at=state.started_at + pause_seconds)
```

```python
def draw_too_small_overlay(window: "curses.window", theme: Theme) -> None:
    """Paint a centered 'terminal too small' overlay. Best-effort: never
    raises on tiny window sizes. Erases the window first."""
    window.erase()
    rows, cols = window.getmaxyx()
    lines = (
        "terminal too small",
        "resize to at least 60x15",
        "esc to abort",
    )
    block_height = len(lines)
    start_y = max(0, (rows - block_height) // 2)
    for i, text in enumerate(lines):
        text = text[:cols]
        x = max(0, (cols - len(text)) // 2)
        y = start_y + i
        if y >= rows:
            break
        try:
            window.addstr(y, x, text, theme.fg_dim)
        except curses.error:
            pass
    try:
        window.refresh()
    except curses.error:
        pass
```

```python
def _init_truecolor_pairs() -> None:
    """Tier 1: terminal supports user-defined RGB. Use exact serika colors."""
    curses.init_color(10, 819, 816, 773)   # fg_correct  #d1d0c5
    curses.init_color(11, 392, 400, 411)   # fg_dim      #646669
    curses.init_color(12, 886, 718, 79)    # fg_caret    #e2b714
    curses.init_color(13, 792, 278, 329)   # fg_error    #ca4754
    curses.init_pair(PAIR_FG_CORRECT, 10, -1)
    curses.init_pair(PAIR_FG_DIM, 11, -1)
    curses.init_pair(PAIR_FG_CARET, 12, -1)
    curses.init_pair(PAIR_FG_ERROR, 13, -1)


def _init_16color_pairs() -> None:
    """Tier 2: 16 colors available but can't redefine. Use bright variants:
    index 8 is the canonical 'bright black' = gray on every modern theme,
    visible on both dark and light backgrounds.
    """
    curses.init_pair(PAIR_FG_CORRECT, curses.COLOR_WHITE, -1)
    curses.init_pair(PAIR_FG_DIM, 8, -1)       # bright black / gray
    curses.init_pair(PAIR_FG_CARET, 11, -1)    # bright yellow
    curses.init_pair(PAIR_FG_ERROR, curses.COLOR_RED, -1)


def _init_8color_pairs() -> None:
    """Tier 3: only the basic 8 colors. Accept loss of typed/untyped contrast
    so dim text remains visible. The caret and red errors still distinguish
    position and mistakes.

    curses.init_pair(PAIR_FG_CORRECT, curses.COLOR_WHITE, -1)
    curses.init_pair(PAIR_FG_DIM, curses.COLOR_WHITE, -1)   # same as correct
    curses.init_pair(PAIR_FG_CARET, curses.COLOR_YELLOW, -1)
    curses.init_pair(PAIR_FG_ERROR, curses.COLOR_RED, -1)


def init_pairs() -> Theme:
    """Initialize curses color pairs and return the Theme that references them.

    Must be called after curses.start_color(). Three tiers:
      1. can_change_color() and COLORS >= 16 — exact serika RGB
      2. COLORS >= 16 — bright-variant fallback (visible dim text)
      3. otherwise — readable 8-color fallback (correct/dim collapse)
    """
    curses.start_color()
    curses.use_default_colors()  # let the terminal background show through

    if curses.can_change_color() and curses.COLORS >= 16:
        _init_truecolor_pairs()
    elif curses.COLORS >= 16:
        _init_16color_pairs()
    else:
        _init_8color_pairs()

    return Theme(
        fg_correct=curses.color_pair(PAIR_FG_CORRECT),
        fg_dim=curses.color_pair(PAIR_FG_DIM),
        fg_caret=curses.color_pair(PAIR_FG_CARET),
        fg_error=curses.color_pair(PAIR_FG_ERROR),
    )
```

```python
def _wait_until_restored(window, theme) -> tuple[str, float]:
    """Block until the terminal is at least MIN_COLS x MIN_ROWS, the user
    aborts (Esc or Ctrl-C), or the process is interrupted.

    Returns:
      ("restored", pause_seconds) — caller resumes (test or result screen).
      ("aborted",  pause_seconds) — caller aborts.

    The pause_seconds value is the wall-clock time spent waiting. Callers
    that own a timer should pass it to apply_pause_offset and shift any
    sample-cadence locals by the same amount.

    During the pause, only Esc / Ctrl-C / KEY_RESIZE are acted on. Other
    keystrokes are dropped so they can't accidentally dismiss the overlay.
    """
    paused_at = time.monotonic()
    render.draw_too_small_overlay(window, theme)
    window.timeout(-1)
    try:
        while True:
            keycode = window.getch()
            if keycode == -1:
                continue
            key = translate(keycode)
            if key.kind is KeyKind.RESIZE:
                rows, cols = window.getmaxyx()
                if not is_too_small(rows, cols):
                    return ("restored", time.monotonic() - paused_at)
                render.draw_too_small_overlay(window, theme)
                continue
            if key.kind in (KeyKind.ESC, KeyKind.CTRL_C):
                return ("aborted", time.monotonic() - paused_at)
            # Any other keypress is ignored during pause.
    finally:
        window.timeout(50)
```

```python
                        if is_too_small(rows, cols):
                            result, _ = _wait_until_restored(window, theme)
                            if result == "aborted":
                                aborted = True
                                action = "exit"
                                break
                        render.draw_result_screen(window, state, summary, label, theme, was_pb=was_pb)
                        continue
```

````markdown
## Stats

slugtype reports four numbers at the end of every test:

- **wpm** — net words per minute: `(correct word chars + correct spaces between fully-correct words) * 60 / test_seconds / 5`
- **raw** — raw words per minute (counts all keystrokes including incorrect)
- **acc** — character-level accuracy: `100 * correct_chars / (correct_chars + incorrect_chars)`
- **consistency** — 1σ stability of per-second raw wpm (higher is more consistent)

## History

Completed tests are appended to a JSONL history file:

- macOS: `~/Library/Application Support/slugtype/history.jsonl`
- Linux: `$XDG_DATA_HOME/slugtype/history.jsonl` (default `~/.local/share/slugtype/history.jsonl`)

Personal-best detection compares against this file. Delete it to reset PBs.

## Keys

| Key | Action |
|---|---|
| Tab, Ctrl-R | Restart with the same config |
| Esc | Open the settings menu (or close the result screen) |
| n | (Result screen) Open settings menu |
| q | (Result screen) Quit |
| Ctrl-C | Abort immediately |
````
