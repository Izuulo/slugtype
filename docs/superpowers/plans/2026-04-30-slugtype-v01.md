**Goal:** Build the v0.1 walking skeleton of `slugtype` — a curses-based typing test that loads `english-200`, runs a 60-second test (default), accepts letters / space / backspace, and prints `wpm: <int>  acc: <float>%` on stdout when the timer expires.

**Architecture:** Pure-logic core (`state.py`, `stats.py`, `words.py`, `input.py`) is fully unit-tested without a TTY. Curses I/O lives in `terminal.py`, `render.py`, `run.py`. `cli.py` only does argparse and dispatches into `run.py`. State transitions are pure: `apply_keypress(state, key, now) -> new_state`.

**Tech Stack:** Python 3.9+, stdlib `curses`, `dataclasses`, `argparse`, `random`. Tests use `pytest`. No runtime deps beyond stdlib.

```python
"""Word-list loading and sampling. Pure given a seeded RNG."""
from __future__ import annotations

import random
from importlib.resources import files


def load(name: str) -> tuple[str, ...]:
    """Load a word list from the bundled data/ directory.

    Returns a tuple of words. Raises FileNotFoundError if the named list
    does not exist.
    """
    data_file = files("slugtype.data").joinpath(f"{name}.txt")
    text = data_file.read_text(encoding="utf-8")
    word_list = tuple(line.strip() for line in text.splitlines() if line.strip())
    if not word_list:
        raise ValueError(f"word list {name!r} is empty")
    return word_list
```

```python
def sample(word_list: tuple[str, ...], n: int, rng: random.Random) -> tuple[str, ...]:
    """Sample n words from word_list with replacement using the provided RNG.

    n may exceed len(word_list); n=0 returns an empty tuple.
    """
    if n <= 0:
        return ()
    return tuple(rng.choice(word_list) for _ in range(n))
```

```python
"""Pure TestState dataclass and transitions. No I/O."""
from __future__ import annotations

from dataclasses import dataclass, replace
from typing import Literal

from slugtype.input import Key

Category = Literal["correct", "incorrect", "extra"]
Mode = Literal["time", "words"]


@dataclass(frozen=True)
class TypedChar:
    expected: str
    typed: str
    category: Category


@dataclass(frozen=True)
class WordEntry:
    expected: str
    typed: tuple[TypedChar, ...]
    submitted: bool


@dataclass(frozen=True)
class TestState:
    words: tuple[WordEntry, ...]
    current_index: int
    started_at: float | None
    ended_at: float | None
    duration_s: float
    mode: Mode
    finished: bool


def initial_state(words: tuple[str, ...], mode: Mode, duration_s: float) -> TestState:
    return TestState(
        words=tuple(WordEntry(expected=w, typed=(), submitted=False) for w in words),
        current_index=0,
        started_at=None,
        ended_at=None,
        duration_s=duration_s,
        mode=mode,
        finished=False,
    )
```

```python
"""Curses keycode → Key enum translation."""
from __future__ import annotations

from dataclasses import dataclass
from enum import Enum, auto


class KeyKind(Enum):
    CHAR = auto()
    SPACE = auto()
    BACKSPACE = auto()
    CTRL_C = auto()
    RESIZE = auto()
    UNKNOWN = auto()


@dataclass(frozen=True)
class Key:
    kind: KeyKind
    char: str = ""  # populated only when kind is CHAR
```

```python
def apply_keypress(state: TestState, key: Key, now: float) -> TestState:
    if state.finished:
        return state
    if key.kind is KeyKind.CHAR:
        return _apply_char(state, key.char, now)
    return state


def _apply_char(state: TestState, ch: str, now: float) -> TestState:
    started_at = state.started_at if state.started_at is not None else now
    current = state.words[state.current_index]
    typed_index = len(current.typed)
    expected = current.expected
    if typed_index < len(expected):
        if ch == expected[typed_index]:
            entry = TypedChar(expected[typed_index], ch, "correct")
        else:
            entry = TypedChar(expected[typed_index], ch, "incorrect")
    else:
        entry = TypedChar("", ch, "extra")
    new_word = WordEntry(
        expected=current.expected,
        typed=current.typed + (entry,),
        submitted=current.submitted,
    )
    new_words = state.words[: state.current_index] + (new_word,) + state.words[state.current_index + 1 :]
    return replace(state, words=new_words, started_at=started_at)
```

```python
def _apply_space(state: TestState, now: float) -> TestState:
    current = state.words[state.current_index]
    if len(current.typed) == 0:
        return state  # no-op at word start
    new_word = WordEntry(
        expected=current.expected,
        typed=current.typed,
        submitted=True,
    )
    new_words = state.words[: state.current_index] + (new_word,) + state.words[state.current_index + 1 :]
    is_last = state.current_index == len(state.words) - 1
    if is_last and state.mode == "words":
        return replace(state, words=new_words, finished=True, ended_at=now)
    if is_last:
        # In time mode, last word submitted — but timer still controls end. Stay on last index.
        return replace(state, words=new_words)
    return replace(state, words=new_words, current_index=state.current_index + 1)
```

```python
def _apply_backspace(state: TestState) -> TestState:
    current = state.words[state.current_index]
    if len(current.typed) == 0:
        return state  # no-op; do not cross word boundary
    new_word = WordEntry(
        expected=current.expected,
        typed=current.typed[:-1],
        submitted=current.submitted,
    )
    new_words = state.words[: state.current_index] + (new_word,) + state.words[state.current_index + 1 :]
    return replace(state, words=new_words)


def tick(state: TestState, now: float) -> TestState:
    if state.finished:
        return state
    if state.mode != "time":
        return state
    if state.started_at is None:
        return state
    if now - state.started_at >= state.duration_s:
        return replace(state, finished=True, ended_at=now)
    return state
```

```python
    incorrect = incorrect_chars(state)

    if total == 0:
        return 100.0
    return 100.0 * correct / total
```

```python
@dataclass(frozen=True)
class Summary:
    wpm: float
    raw_wpm: float
    accuracy: float
    correct_chars: int
    incorrect_chars: int
    extra_chars: int


def _is_fully_correct(word: WordEntry) -> bool:
    if not word.submitted:
        return False
    if len(word.typed) != len(word.expected):
        return False
    return all(tc.category == "correct" for tc in word.typed)


def _test_seconds(state: TestState) -> float:
    if state.started_at is None or state.ended_at is None:
        return 0.0
    return state.ended_at - state.started_at


def wpm(state: TestState) -> float:
    seconds = _test_seconds(state)
    if seconds <= 0:
        return 0.0
    fully_correct = [_is_fully_correct(w) for w in state.words]
    correct_word_chars = sum(
        len(w.expected) for w, ok in zip(state.words, fully_correct) if ok
    )
    correct_spaces = 0
    for i in range(len(fully_correct) - 1):
        if fully_correct[i] and fully_correct[i + 1]:
            correct_spaces += 1
    return (correct_word_chars + correct_spaces) * 60.0 / seconds / 5.0


def raw_wpm(state: TestState) -> float:
    seconds = _test_seconds(state)
    if seconds <= 0:
        return 0.0
    typed_chars = sum(len(w.typed) for w in state.words)
    submitted_spaces = sum(1 for w in state.words if w.submitted)
    return (typed_chars + submitted_spaces) * 60.0 / seconds / 5.0


def summary(state: TestState) -> Summary:
    return Summary(
        wpm=wpm(state),
        raw_wpm=raw_wpm(state),
        accuracy=accuracy(state),
        correct_chars=correct_chars(state),
        incorrect_chars=incorrect_chars(state),
        extra_chars=extra_chars(state),
    )
```

```python
import curses


def translate(keycode: int) -> Key:
    if keycode == -1:
        return Key(kind=KeyKind.UNKNOWN)
    if keycode == 3:  # Ctrl-C
        return Key(kind=KeyKind.CTRL_C)
    if keycode == curses.KEY_RESIZE:
        return Key(kind=KeyKind.RESIZE)
    if keycode in (curses.KEY_BACKSPACE, 127, 8):
        return Key(kind=KeyKind.BACKSPACE)
    if keycode == ord(" "):
        return Key(kind=KeyKind.SPACE)
    if 0 <= keycode < 0x110000:
        try:
            ch = chr(keycode)
        except ValueError:
            return Key(kind=KeyKind.UNKNOWN)
        if ch.isprintable() and len(ch) == 1 and not ch.isspace():
            return Key(kind=KeyKind.CHAR, char=ch)
    return Key(kind=KeyKind.UNKNOWN)
```

```python
"""Curses session lifecycle: alt-screen, raw mode, restore on exit."""
from __future__ import annotations

import curses
import sys
from contextlib import contextmanager
from typing import Iterator


MIN_COLS = 60
MIN_ROWS = 15


class TerminalTooSmallError(RuntimeError):
    pass


@contextmanager
def session() -> Iterator["curses.window"]:
    """Enter alt-screen + raw mode for the duration of the block.

    Raises TerminalTooSmallError if the terminal is below MIN_COLS x MIN_ROWS
    when entering. On exit (normal or exceptional) restores the terminal.
    """
    stdscr = curses.initscr()
    try:
        curses.noecho()
        curses.cbreak()
        curses.raw()
        stdscr.keypad(True)
        try:
            curses.curs_set(0)
        except curses.error:
            pass  # cursor visibility not supported

        rows, cols = stdscr.getmaxyx()
        if cols < MIN_COLS or rows < MIN_ROWS:
            raise TerminalTooSmallError(
                f"slugtype: needs at least {MIN_COLS}x{MIN_ROWS}, got {cols}x{rows}"
            )

        yield stdscr
    finally:
        try:
            curses.curs_set(1)
        except curses.error:
            pass
        stdscr.keypad(False)
        curses.noraw()
        curses.nocbreak()
        curses.echo()
        curses.endwin()
```

```python
"""Curses rendering. Pure(ish) over a curses window — no global state."""
from __future__ import annotations

import curses
import math

from slugtype.state import TestState
from slugtype.theme import Theme


def draw(window: "curses.window", state: TestState, theme: Theme, config_label: str, now: float) -> None:
    window.erase()
    draw_top_bar(window, config_label, theme)
    draw_word_stream(window, state, theme)
    draw_bottom_bar(window, state, theme, now)
    window.refresh()


def draw_top_bar(window: "curses.window", config_label: str, theme: Theme) -> None:
    rows, cols = window.getmaxyx()
    left = "slugtype"
    right = config_label
    pad = max(1, cols - len(left) - len(right) - 2)
    text = f" {left}{' ' * pad}{right} "
    text = text[:cols]
    window.addstr(0, 0, text, theme.fg_dim)


def draw_word_stream(window: "curses.window", state: TestState, theme: Theme) -> None:
    """Render the current word and the next two lines of upcoming words.

    The current word is highlighted character by character based on the
    user's typed buffer; upcoming words render dim; the caret is drawn at
    the typing cursor inside the current word.
    """
    rows, cols = window.getmaxyx()
    inner_cols = cols - 4  # 2-col left/right margin
    if inner_cols < 10:
        return  # too narrow to render anything meaningful

    # Build a flat list of (text, attr) segments for the current word and all upcoming words.
    segments: list[tuple[str, int, bool]] = []  # (text, attr, is_caret_target)

    for idx, word in enumerate(state.words):
        if word.submitted and idx < state.current_index:
            continue  # already-completed words scroll out of view
        if idx == state.current_index:
            # Render typed chars with their categorical color.
            for tc in word.typed:
                if tc.category == "correct":
                    segments.append((tc.typed, theme.fg_correct, False))
                elif tc.category == "incorrect":
                    segments.append((tc.expected or tc.typed, theme.fg_error, False))
                else:  # extra
                    segments.append((tc.typed, theme.fg_error | curses.A_DIM, False))
            # Caret marker (one segment so we know where to flash a block).
            untyped_start = len(word.typed)
            untyped_correct = word.expected[untyped_start:]
            if untyped_correct:
                # First untyped char gets the caret highlight; rest is dim foreground.
                segments.append((untyped_correct[0], theme.fg_caret | curses.A_REVERSE, True))
                if len(untyped_correct) > 1:
                    segments.append((untyped_correct[1:], theme.fg_correct | curses.A_DIM, False))
            segments.append((" ", theme.fg_dim, False))
        else:
            segments.append((word.expected, theme.fg_dim, False))
            segments.append((" ", theme.fg_dim, False))

    # Layout segments into 3 lines of `inner_cols` width, breaking on spaces.
    lines: list[list[tuple[str, int]]] = [[]]
    line_lens = [0]
    for text, attr, _ in segments:
        # If adding this segment overflows, soft-wrap on the boundary.
        if line_lens[-1] + len(text) > inner_cols and len(lines) < 3:
            lines.append([])
            line_lens.append(0)
            if text == " ":
                continue  # don't start a new line with a space
        if len(lines) > 3:
            break
        lines[-1].append((text, attr))
        line_lens[-1] += len(text)

    # Draw three lines starting at row 2.
    for i, segs in enumerate(lines[:3]):
        x = 2
        y = 2 + i
        for text, attr in segs:
            try:
                window.addstr(y, x, text, attr)
            except curses.error:
                pass  # writing to bottom-right corner can fail; ignore
            x += len(text)


def draw_bottom_bar(window: "curses.window", state: TestState, theme: Theme, now: float) -> None:
    rows, cols = window.getmaxyx()
    y = rows - 2

    if state.started_at is None:
        left = format_progress(state, elapsed=0.0)
        center = "press any key to start"
    else:
        end = state.ended_at if state.ended_at is not None else now
        elapsed = end - state.started_at
        left = format_progress(state, elapsed=elapsed)
        center = ""

    pad_right = max(0, cols - len(left) - len(center) - 4)
    text = f" {left}  {center}{' ' * pad_right} "
    text = text[:cols]
    window.addstr(y, 0, text, theme.fg_dim)


def format_progress(state: TestState, elapsed: float) -> str:
    if state.mode == "time":
        remaining = max(0.0, state.duration_s - elapsed)
        m = int(remaining // 60)
        s = int(remaining % 60)
        return f"{m}:{s:02d}"
    # words mode: how many words submitted so far
    submitted = sum(1 for w in state.words if w.submitted)
    return f"{submitted}/{len(state.words)}"
```

Notes for the implementer:
- `draw_word_stream` is intentionally ~50 lines. Keep this as one function for v0.1; if it grows, split out word-segment building from layout. Don't pre-emptively refactor.
- `render.draw` takes `now` so the bottom-bar countdown updates on every redraw. The run loop forces a redraw on every tick (every ~1s) so the countdown visibly decrements.

```python
"""High-level test orchestration: setup terminal, run loop, print result."""
from __future__ import annotations

import curses
import random
import time
from dataclasses import dataclass
from typing import Optional

from slugtype import render, stats, words
from slugtype.input import KeyKind, translate
from slugtype.state import TestState, apply_keypress, initial_state, tick
from slugtype.terminal import TerminalTooSmallError, session
from slugtype.theme import init_pairs


@dataclass(frozen=True)
class Config:
    mode: str           # "time" or "words"
    duration_s: float   # used when mode == "time"
    word_count: int     # used when mode == "words"
    language: str       # "english-200" at v0.1


SAMPLE_OVERSHOOT = 5  # extra word buffer beyond expected typing speed


def _config_label(config: Config) -> str:
    if config.mode == "time":
        return f"{int(config.duration_s)}s · {config.language}"
    return f"{config.word_count} words · {config.language}"


def _build_initial_state(config: Config, rng: random.Random) -> TestState:
    word_pool = words.load(config.language)
    if config.mode == "time":
        # Generate enough words to cover even a fast typist: ~250wpm * (duration/60) words.
        target = max(50, int(250 * config.duration_s / 60))
        sample = words.sample(word_pool, target, rng)
        return initial_state(sample, "time", config.duration_s)
    sample = words.sample(word_pool, config.word_count, rng)
    return initial_state(sample, "words", duration_s=0.0)


def run_test(config: Config) -> Optional[stats.Summary]:
    """Run a single test. Returns the summary on natural completion, or None
    if the user aborted (Ctrl-C).
    """
    rng = random.Random()
    state = _build_initial_state(config, rng)
    label = _config_label(config)

    aborted = False
    final_state = state

    try:
        with session() as window:
            theme = init_pairs()
            window.timeout(50)  # ms; getch blocks up to this long
            last_tick = 0.0
            render.draw(window, state, theme, label, time.monotonic())

            while True:
                now = time.monotonic()

                keycode = window.getch()
                if keycode != -1:
                    key = translate(keycode)
                    if key.kind is KeyKind.CTRL_C:
                        aborted = True
                        break
                    if key.kind is KeyKind.RESIZE:
                        render.draw(window, state, theme, label, now)
                        continue
                    new_state = apply_keypress(state, key, now)
                    if new_state is not state:
                        state = new_state
                        render.draw(window, state, theme, label, now)

                if state.started_at is not None and (now - last_tick) >= 1.0:
                    last_tick = now
                    new_state = tick(state, now)
                    state = new_state  # may or may not be the same object; redraw either way to update countdown
                    render.draw(window, state, theme, label, now)

                if state.finished:
                    final_state = state
                    break
    except KeyboardInterrupt:
        aborted = True
    except TerminalTooSmallError as e:
        # Re-raise so cli.py can print to stderr after teardown.
        raise

    if aborted:
        return None
    return stats.summary(final_state)
```

```python
"""Argparse and program entry point. No curses imports here."""
from __future__ import annotations

import argparse
import sys
from typing import Optional, Sequence

from slugtype import __version__
from slugtype.run import Config, run_test
from slugtype.terminal import TerminalTooSmallError


VALID_TIMES = (15, 30, 60, 120)
VALID_WORDS = (10, 25, 50, 100)
DEFAULT_TIME = 60


def _build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(prog="slugtype", description="A terminal typing test.")
    group = parser.add_mutually_exclusive_group()
    group.add_argument(
        "--time",
        type=int,
        choices=VALID_TIMES,
        help=f"time-mode duration in seconds (one of {VALID_TIMES}); default {DEFAULT_TIME}",
    )
    group.add_argument(
        "--words",
        type=int,
        choices=VALID_WORDS,
        help=f"words-mode target (one of {VALID_WORDS})",
    )
    parser.add_argument("--version", action="version", version=f"slugtype {__version__}")
    return parser


def _config_from_args(args: argparse.Namespace) -> Config:
    if args.words is not None:
        return Config(mode="words", duration_s=0.0, word_count=args.words, language="english-200")
    duration = args.time if args.time is not None else DEFAULT_TIME
    return Config(mode="time", duration_s=float(duration), word_count=0, language="english-200")


def main(argv: Optional[Sequence[str]] = None) -> int:
    parser = _build_parser()
    args = parser.parse_args(argv)
    config = _config_from_args(args)

    try:
        summary = run_test(config)
    except TerminalTooSmallError as e:
        print(str(e), file=sys.stderr)
        return 1
    except FileNotFoundError as e:
        print(f"slugtype: word list missing: {e}", file=sys.stderr)
        return 1
    except ValueError as e:
        print(f"slugtype: {e}", file=sys.stderr)
        return 1
    except Exception as e:
        # Catches curses.error (TERM not set / TERM=dumb) and other unexpected
        # terminal-init failures. Print a clear message and exit non-zero
        # rather than leaking a traceback into a half-restored terminal.
        print(f"slugtype: terminal error: {e}", file=sys.stderr)
        return 1

    if summary is None:
        return 0  # aborted; nothing to print

    print(f"wpm: {round(summary.wpm)}  acc: {summary.accuracy:.1f}%")
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```
