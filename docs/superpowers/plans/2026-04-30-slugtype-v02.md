# slugtype v0.2 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Extend slugtype from v0.1 (walking skeleton) to v0.2 (full test loop): live wpm in the bottom bar, four-stat result screen with consistency, per-second sampling, tab restart, esc menu for config changes, bundled english-1k list, `--language` and `--no-live-wpm` CLI flags.

**Architecture:** All new logic stays in pure modules where possible (`stats.consistency`, `stats.live_wpm`, `state.append_sample`). Per-second sampling lives in the run loop (it owns the wall clock); state stores the resulting samples. The menu is local UI state in the run loop — no separate module yet, since that would over-engineer for v0.2.

**Tech Stack:** Python 3.9+, stdlib `curses`, stdlib only at runtime. `pytest` for tests.

**Specs:**
- PRD: [`docs/2026-04-30-initial-prd.md`](../../2026-04-30-initial-prd.md) — §5 UX, §6 stat formulas, §13 milestones

**Decisions for open questions (PRD §15):**
- Live wpm default: ON (matches Monkeytype). `--no-live-wpm` hides it.
- Tab restart key: yes, mirror Monkeytype.
- Quiet mode: out of scope for v0.2; defer to v1.1+.

```python
import math


def consistency(state: TestState) -> float:
    """Kogasa consistency: 100 * (1 - tanh(cv + cv³/3)) where cv = stdev/mean
    over the per-second raw-WPM samples.
    Returns 100.0 when there are fewer than 2 samples or the mean is 0.
    """
    samples = state.raw_per_second
    if len(samples) < 2:
        return 100.0
    mean = sum(samples) / len(samples)
    if mean == 0:
        return 100.0
    variance = sum((x - mean) ** 2 for x in samples) / len(samples)
    stdev = math.sqrt(variance)
    cv = stdev / mean
    return 100.0 * (1 - math.tanh(cv + (cv ** 3) / 3))
```

```python
def live_wpm(state: TestState, now: float) -> float:
    """Current WPM using elapsed = now - started_at as the denominator.
    Same fully-correct-word logic as wpm(). Returns 0.0 if not started.
    """
    if state.started_at is None:
        return 0.0
    elapsed = now - state.started_at
    if elapsed <= 0:
        return 0.0
    fully_correct = [_is_fully_correct(w) for w in state.words]
    correct_word_chars = sum(
        len(w.expected) for w, ok in zip(state.words, fully_correct) if ok
    )
    correct_spaces = 0
    for i in range(len(fully_correct) - 1):
        if fully_correct[i] and fully_correct[i + 1]:
            correct_spaces += 1
    return (correct_word_chars + correct_spaces) * 60.0 / elapsed / 5.0
```

```python
def translate(keycode: int) -> Key:
    if keycode == -1:
        return Key(kind=KeyKind.UNKNOWN)
    if keycode == 3:
        return Key(kind=KeyKind.CTRL_C)
    if keycode == 9:
        return Key(kind=KeyKind.TAB)
    if keycode == 27:
        return Key(kind=KeyKind.ESC)
    if keycode in (10, 13, curses.KEY_ENTER):
        return Key(kind=KeyKind.ENTER)
    if keycode == curses.KEY_UP:
        return Key(kind=KeyKind.UP)
    if keycode == curses.KEY_DOWN:
        return Key(kind=KeyKind.DOWN)
    if keycode == curses.KEY_LEFT:
        return Key(kind=KeyKind.LEFT)
    if keycode == curses.KEY_RIGHT:
        return Key(kind=KeyKind.RIGHT)
    if keycode == curses.KEY_RESIZE:
        return Key(kind=KeyKind.RESIZE)
    if keycode in (curses.KEY_BACKSPACE, 127, 8):
        return Key(kind=KeyKind.BACKSPACE)
    if keycode == ord(" "):
        return Key(kind=KeyKind.SPACE)
    if 0 <= keycode < 128:
        try:
            ch = chr(keycode)
        except ValueError:
            return Key(kind=KeyKind.UNKNOWN)
        if ch.isprintable() and len(ch) == 1 and not ch.isspace():
            return Key(kind=KeyKind.CHAR, char=ch)
    return Key(kind=KeyKind.UNKNOWN)
```

```python
"""Fetch word lists from upstream and write to src/slugtype/data/.

Dev tool only — not shipped at runtime. Run from the repo root:

    python scripts/fetch_word_lists.py
"""
from __future__ import annotations

import json
import sys
import urllib.request
from pathlib import Path


SOURCES = {
    "english-1k": "https://raw.githubusercontent.com/monkeytypegame/monkeytype/master/frontend/static/languages/english_1k.json",
}

DEST = Path(__file__).resolve().parent.parent / "src" / "slugtype" / "data"


def fetch_and_write(name: str, url: str) -> int:
    print(f"Fetching {name} from {url}", file=sys.stderr)
    req = urllib.request.Request(url, headers={"User-Agent": "slugtype-dev/0.2"})
    with urllib.request.urlopen(req, timeout=30) as resp:
        payload = json.load(resp)
    raw_words = payload.get("words", [])
    # Filter to lowercase ASCII alphabetic words; preserve order.
    words = []
    seen = set()
    for w in raw_words:
        normalised = w.lower()
        if not normalised.isascii() or not normalised.isalpha():
            continue
        if normalised in seen:
            continue
        seen.add(normalised)
        words.append(normalised)
    DEST.mkdir(parents=True, exist_ok=True)
    out_path = DEST / f"{name}.txt"
    out_path.write_text("\n".join(words) + "\n", encoding="utf-8")
    print(f"Wrote {len(words)} words to {out_path}", file=sys.stderr)
    return len(words)


def main() -> int:
    for name, url in SOURCES.items():
        fetch_and_write(name, url)
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

```python
def draw_menu(
    window: "curses.window",
    items: list[tuple[str, str]],
    selected_index: int,
    theme: Theme,
) -> None:
    """Centered modal listing menu items as `label  value` rows.

    `items` is a list of (label, current_value_string) pairs. `selected_index`
    is the row to highlight.
    """
    rows, cols = window.getmaxyx()

    inner_width = max(28, max((len(lbl) + len(val) for lbl, val in items), default=24) + 8)
    box_width = min(cols - 4, inner_width + 4)
    box_height = len(items) + 4
    box_top = max(0, (rows - box_height) // 2)
    box_left = max(0, (cols - box_width) // 2)

    # Box border
    horiz = "─" * (box_width - 2)
    try:
        window.addstr(box_top, box_left, "┌" + horiz + "┐", theme.fg_dim)
        for i in range(1, box_height - 1):
            window.addstr(box_top + i, box_left, "│" + " " * (box_width - 2) + "│", theme.fg_dim)
        window.addstr(box_top + box_height - 1, box_left, "└" + horiz + "┘", theme.fg_dim)
    except curses.error:
        pass

    # Title
    title = "settings"
    try:
        window.addstr(box_top + 1, box_left + (box_width - len(title)) // 2, title, theme.fg_dim)
    except curses.error:
        pass

    # Items
    for i, (label, value) in enumerate(items):
        y = box_top + 3 + i
        attr_label = theme.fg_caret if i == selected_index else theme.fg_correct
        attr_value = theme.fg_caret | curses.A_REVERSE if i == selected_index else theme.fg_dim
        try:
            window.addstr(y, box_left + 3, label, attr_label)
            value_x = box_left + box_width - 3 - len(value)
            window.addstr(y, value_x, value, attr_value)
        except curses.error:
            pass

    window.refresh()
```

```python
TIME_CHOICES = (15, 30, 60, 120)
WORDS_CHOICES = (10, 25, 50, 100)

@dataclass
class MenuState:
    mode: str
    time_length: int
    words_count: int
    language: str
    show_live_wpm: bool
    selected: int = 0


def _menu_state_from_config(config: Config) -> MenuState:
    return MenuState(
        mode=config.mode,
        time_length=int(config.duration_s) if config.mode == "time" else TIME_CHOICES[2],
        words_count=config.word_count if config.mode == "words" else WORDS_CHOICES[1],
        language=config.language,
        show_live_wpm=config.show_live_wpm,
    )


def _config_from_menu(menu: MenuState) -> Config:
    if menu.mode == "time":
        return Config(
            mode="time",
            duration_s=float(menu.time_length),
            word_count=0,
            language=menu.language,
            show_live_wpm=menu.show_live_wpm,
        )
    return Config(
        mode="words",
        duration_s=0.0,
        word_count=menu.words_count,
        language=menu.language,
        show_live_wpm=menu.show_live_wpm,
    )


def _menu_items(menu: MenuState) -> list[tuple[str, str]]:
    if menu.mode == "time":
        length_label = "duration"
        length_value = f"{menu.time_length}s"
    else:
        length_label = "words"
        length_value = str(menu.words_count)
    return [
        ("mode", menu.mode),
        (length_label, length_value),
        ("language", menu.language),
        ("live wpm", "on" if menu.show_live_wpm else "off"),
    ]


def _cycle(values, current, direction: int):
    if current not in values:
        return values[0]
    idx = values.index(current)
    return values[(idx + direction) % len(values)]


def _menu_apply_arrow(menu: MenuState, key) -> MenuState:
    if key.kind is KeyKind.UP:
        menu.selected = (menu.selected - 1) % 4
    elif key.kind is KeyKind.DOWN:
        menu.selected = (menu.selected + 1) % 4
    elif key.kind in (KeyKind.LEFT, KeyKind.RIGHT):
        direction = -1 if key.kind is KeyKind.LEFT else 1
        if menu.selected == 0:
            menu.mode = _cycle(("time", "words"), menu.mode, direction)
        elif menu.selected == 1:
            if menu.mode == "time":
                menu.time_length = _cycle(TIME_CHOICES, menu.time_length, direction)
            else:
                menu.words_count = _cycle(WORDS_CHOICES, menu.words_count, direction)
        elif menu.selected == 2:
            menu.language = _cycle(LANGUAGES, menu.language, direction)
        elif menu.selected == 3:
            menu.show_live_wpm = not menu.show_live_wpm
    return menu
```

```python
"""Render layer tests using a window-shaped recorder fake. No TTY required."""
from dataclasses import replace

from slugtype import render
from slugtype.state import initial_state
from slugtype.theme import Theme


class WindowRecorder:
    """Record addstr calls so tests can assert on layout without a real curses window."""

    def __init__(self, rows: int = 24, cols: int = 80):
        self.rows = rows
        self.cols = cols
        self.calls: list[tuple[int, int, str, int]] = []

    def getmaxyx(self):
        return (self.rows, self.cols)

    def addstr(self, y, x, text, attr=0):
        self.calls.append((y, x, text, attr))

    def erase(self):
        self.calls.clear()

    def refresh(self):
        pass

    def lines(self) -> dict[int, str]:
        """Reconstruct an approximate per-row string view from the recorded calls."""
        out: dict[int, list[str]] = {}
        for y, x, text, _ in self.calls:
            row = out.setdefault(y, [" "] * self.cols)
            for i, ch in enumerate(text):
                if 0 <= x + i < self.cols:
                    row[x + i] = ch
        return {y: "".join(row).rstrip() for y, row in out.items()}


# Stub theme — attribute values are arbitrary integers, just used as opaque identifiers.
THEME = Theme(fg_correct=1, fg_dim=2, fg_caret=3, fg_error=4)


def test_bottom_bar_idle_shows_press_any_key_hint():
    s = initial_state(("the",), "time", 60.0)
    win = WindowRecorder(rows=24, cols=80)
    render.draw_bottom_bar(win, s, THEME, now=0.0, live_wpm_value=None)
    line = win.lines()[22]  # rows-2 = 22 for a 24-row window
    assert "1:00" in line
    assert "press any key to start" in line


def test_bottom_bar_running_with_live_wpm():
    s = initial_state(("the",), "time", 60.0)
    s = replace(s, started_at=0.0)
    win = WindowRecorder(rows=24, cols=80)
    render.draw_bottom_bar(win, s, THEME, now=10.0, live_wpm_value=42.0)
    line = win.lines()[22]
    assert "0:50" in line
    assert "42 wpm" in line


def test_bottom_bar_running_without_live_wpm_when_disabled():
    s = initial_state(("the",), "time", 60.0)
    s = replace(s, started_at=0.0)
    win = WindowRecorder(rows=24, cols=80)
    render.draw_bottom_bar(win, s, THEME, now=10.0, live_wpm_value=None)
    line = win.lines()[22]
    assert "0:50" in line
    assert "wpm" not in line


def test_menu_renders_all_items_and_highlights_selected():
    items = [("mode", "time"), ("duration", "60s"), ("language", "english-1k"), ("live wpm", "on")]
    win = WindowRecorder(rows=24, cols=80)
    render.draw_menu(win, items, selected_index=1, theme=THEME)
    rendered = "\n".join(win.lines().values())
    for label, value in items:
        assert label in rendered
        assert value in rendered
    # The selected row's value attr should be the reversed caret attr; verify at least
    # one call carries that attribute (smoke-level check).
    assert any(attr & THEME.fg_caret for _, _, _, attr in win.calls)
```
