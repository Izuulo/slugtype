```python
"""Platform-aware data directory resolution. No I/O at import time."""
from __future__ import annotations

import os
import sys
from pathlib import Path


def data_dir() -> Path:
    """Return the slugtype data directory, creating it if missing.

    macOS: ~/Library/Application Support/slugtype/
    Linux: $XDG_DATA_HOME/slugtype/ or ~/.local/share/slugtype/

    If directory creation fails (read-only fs, permission denied), the path
    is still returned; downstream callers swallow write errors.
    """
    if sys.platform == "darwin":
        base = Path.home() / "Library" / "Application Support"
    else:
        xdg = os.environ.get("XDG_DATA_HOME")
        base = Path(xdg) if xdg else Path.home() / ".local" / "share"
    path = base / "slugtype"
    try:
        path.mkdir(parents=True, exist_ok=True)
    except OSError:
        pass
    return path


def history_path() -> Path:
    """Return the path to history.jsonl. Resolves data_dir() each call so
    tests can monkeypatch it without import-time side effects.
    """
    return data_dir() / "history.jsonl"
```

```python
"""History persistence and personal-best queries. JSONL-backed."""
from __future__ import annotations

import json
from dataclasses import dataclass
from datetime import datetime
from pathlib import Path
from typing import Optional, Tuple

from slugtype.state import TestState

ConfigKey = Tuple[str, int, str, bool, bool]


@dataclass(frozen=True)
class Record:
    ts: str                          # ISO 8601 UTC, e.g. "2026-05-07T19:14:02Z"
    wpm: float
    raw: float
    acc: float
    consistency: float
    mode: str                        # "time" | "words"
    length: int                      # duration_s for time mode, word_count for words
    language: str
    punctuation: bool                # always False in v0.3a; reserved for v1.0
    numbers: bool                    # always False in v0.3a; reserved for v1.0
    chars: dict
    duration_s: float
    samples: tuple = ()              # per-second raw WPM


def _config_length(config) -> int:
    if config.mode == "time":
        return int(config.duration_s)
    return int(config.word_count)


def record_from_state(state: TestState, summary, config, ts: datetime) -> Record:
    """Build a Record from a finished TestState + Summary + Config + timestamp."""
    duration = (
        (state.ended_at - state.started_at)
        if state.started_at is not None and state.ended_at is not None
        else 0.0
    )
    return Record(
        ts=ts.strftime("%Y-%m-%dT%H:%M:%SZ"),
        wpm=summary.wpm,
        raw=summary.raw_wpm,
        acc=summary.accuracy,
        consistency=summary.consistency,
        mode=config.mode,
        length=_config_length(config),
        language=config.language,
        punctuation=getattr(config, "punctuation", False),
        numbers=getattr(config, "numbers", False),
        chars={
            "correct": summary.correct_chars,
            "incorrect": summary.incorrect_chars,
            "extra": summary.extra_chars,
            "missed": 0,
        },
        duration_s=duration,
        samples=tuple(state.raw_per_second),
    )
```

```python
def config_key(record_or_config) -> ConfigKey:
    """Build the PB key tuple from a Record or Config. Tuple shape:
    (mode, length, language, punctuation, numbers).
    """
    if isinstance(record_or_config, Record):
        r = record_or_config
        return (r.mode, r.length, r.language, r.punctuation, r.numbers)
    c = record_or_config
    return (
        c.mode,
        _config_length(c),
        c.language,
        getattr(c, "punctuation", False),
        getattr(c, "numbers", False),
    )


def _record_to_json(record: Record) -> dict:
    return {
        "ts": record.ts,
        "wpm": record.wpm,
        "raw": record.raw,
        "acc": record.acc,
        "consistency": record.consistency,
        "mode": record.mode,
        "length": record.length,
        "language": record.language,
        "punctuation": record.punctuation,
        "numbers": record.numbers,
        "chars": record.chars,
        "duration_s": record.duration_s,
        "samples": list(record.samples),
    }


def _record_from_json(d: dict) -> Optional[Record]:
    """Build a Record from a JSON dict; return None if required fields are
    missing or malformed. Optional fields default to safe values.
    """
    try:
        return Record(
            ts=str(d["ts"]),
            wpm=float(d["wpm"]),
            raw=float(d.get("raw", 0.0)),
            acc=float(d["acc"]),
            consistency=float(d.get("consistency", 100.0)),
            mode=str(d["mode"]),
            length=int(d["length"]),
            language=str(d["language"]),
            punctuation=bool(d.get("punctuation", False)),
            numbers=bool(d.get("numbers", False)),
            chars=d.get("chars", {"correct": 0, "incorrect": 0, "extra": 0, "missed": 0}),
            duration_s=float(d.get("duration_s", 0.0)),
            samples=tuple(d.get("samples", ())),
        )
    except (KeyError, ValueError, TypeError):
        return None


def append(record: Record, path: Path) -> None:
    """Best-effort: write one JSON line to `path`. Swallows OSError so a
    finished test always reaches the result screen.
    """
    try:
        path.parent.mkdir(parents=True, exist_ok=True)
        with open(path, "a") as f:
            json.dump(_record_to_json(record), f)
            f.write("\n")
    except OSError:
        pass


def load_all(path: Path) -> list[Record]:
    """Load all records from `path`. Returns [] for missing file. Skips
    malformed JSON lines individually.
    """
    if not path.exists():
        return []
    records: list[Record] = []
    try:
        with open(path) as f:
            for line in f:
                line = line.strip()
                if not line:
                    continue
                try:
                    d = json.loads(line)
                except json.JSONDecodeError:
                    continue
                rec = _record_from_json(d)
                if rec is not None:
                    records.append(rec)
    except OSError:
        return []
    return records
```

```python
def pb_qualifies(state: TestState, summary) -> bool:
    """A run qualifies for PB consideration if it ran ≥ 1s and accuracy ≥ 75%."""
    duration = (
        (state.ended_at - state.started_at)
        if state.started_at is not None and state.ended_at is not None
        else 0.0
    )
    if duration < 1.0:
        return False
    if summary.accuracy < 75.0:
        return False
    return True


def pb_for(records: list[Record], key: ConfigKey) -> Optional[Record]:
    """Return the highest-wpm record matching `key` and acc >= 75. None if no match.
    The acc filter is defense in depth: even if a low-acc record landed in the
    file (e.g. from an older format), it must never be crowned as PB.
    """
    matching = [r for r in records if config_key(r) == key and r.acc >= 75.0]
    if not matching:
        return None
    return max(matching, key=lambda r: r.wpm)
```

```python
def chart_lines(samples: Sequence[float], width: int, plot_height: int = 4) -> list[str]:
    if len(samples) < 2 or width < 10 or plot_height < 2:
        return []

    plot_width = width - LABEL_WIDTH - 1  # -1 for the "┤"/"┼" edge column
    if plot_width < 4:
        return []

    # Y-axis range: 0 to ceil(max/15)*15, with a floor of 15.
    raw_max = max(samples)
    y_top = max(15, int(math.ceil(raw_max / 15) * 15))

    # Map each plot column to a sample index, then to a row (0=top, plot_height-1=bottom).
    n = len(samples)
    col_rows = []
    for c in range(plot_width):
        idx = round(c * (n - 1) / max(1, plot_width - 1))
        v = samples[idx]
        r = (plot_height - 1) - round(v / y_top * (plot_height - 1))
        # Clamp defensively.
        r = max(0, min(plot_height - 1, r))
        col_rows.append(r)

    # Build a 2D grid of plot characters.
    grid = [[" "] * plot_width for _ in range(plot_height)]
    for c in range(plot_width):
        r = col_rows[c]
        if c == 0:
            grid[r][c] = "─"
            continue
        prev_r = col_rows[c - 1]
        if r == prev_r:
            grid[r][c] = "─"
        elif r < prev_r:
            # Going up. Previous col turns up (╯), this col starts the new level (╭).
            if grid[prev_r][c - 1] in (" ", "─"):
                grid[prev_r][c - 1] = "╯"
            grid[r][c] = "╭"
            for rr in range(r + 1, prev_r):
                if grid[rr][c - 1] == " ":
                    grid[rr][c - 1] = "│"
        else:
            # Going down. Previous col turns down (╮), this col starts new level (╰).
            if grid[prev_r][c - 1] in (" ", "─"):
                grid[prev_r][c - 1] = "╮"
            grid[r][c] = "╰"
            for rr in range(prev_r + 1, r):
                if grid[rr][c - 1] == " ":
                    grid[rr][c - 1] = "│"

    # Y-axis labels: top row = y_top, axis row = 0, plot rows in between.
    plot_lines: list[str] = []
    for r in range(plot_height):
        label_value = round(y_top * (plot_height - 1 - r) / (plot_height - 1))
        label = f"{label_value:>3} ┤"
        plot_lines.append(label + "".join(grid[r]))

    axis_line = "  0 ┼" + "─" * plot_width

    # X-tick row: 5 ticks evenly spaced. Sample-per-second assumed → max=n.
    tick_chars = [" "] * width
    tick_positions = [0, plot_width // 4, plot_width // 2, 3 * plot_width // 4, plot_width - 1]
    duration = n
    for pos in tick_positions:
        tick_value = round(pos * duration / max(1, plot_width - 1))
        s = f"{tick_value}s"
        start = LABEL_WIDTH + 1 + pos
        for i, ch in enumerate(s):
            target = start + i
            if 0 <= target < width:
                tick_chars[target] = ch
    tick_line = "".join(tick_chars)

    # Pad/truncate every line to exactly `width`.
    out = []
    for line in plot_lines + [axis_line, tick_line]:
        if len(line) < width:
            line = line + " " * (width - len(line))
        elif len(line) > width:
            line = line[:width]
        out.append(line)
    return out
```

```python
def _draw_sparkline(window: "curses.window", samples, x: int, y: int, width: int, plot_height: int, theme: Theme) -> int:
    """Render a sparkline at (y, x). Returns the number of rows written
    (0 if the chart was omitted because chart_lines returned []).
    """
    rows_text = sparkline.chart_lines(samples, width=width, plot_height=plot_height)
    if not rows_text:
        return 0
    plot_chars = set("─╭╮╯╰│┤┼")
    for i, line in enumerate(rows_text):
        # Color the plot characters with fg_correct, axis/labels with fg_dim.
        col = x
        for ch in line:
            attr = theme.fg_correct if ch in plot_chars and ch not in "┤┼" else theme.fg_dim
            try:
                window.addstr(y + i, col, ch, attr)
            except curses.error:
                pass
            col += 1
    return len(rows_text)
```

```python
def _finalize(
    state: TestState,
    summary: stats.Summary,
    config: Config,
    now_utc: datetime,
    history_file,
) -> bool:
    """Persist the finished test (if it qualifies) and decide whether it's
    a personal best for its config. Returns `was_pb`.

    `history_file` is a Path; injected so tests can substitute a tmp path.
    """
    if not history.pb_qualifies(state, summary):
        return False

    record = history.record_from_state(state, summary, config, now_utc)
    # Read existing records *before* appending so the new run isn't its own baseline.
    prior = history.load_all(history_file)
    key = history.config_key(record)
    prev_pb = history.pb_for(prior, key)
    was_pb = prev_pb is None or record.wpm > prev_pb.wpm

    history.append(record, history_file)
    return was_pb
```
